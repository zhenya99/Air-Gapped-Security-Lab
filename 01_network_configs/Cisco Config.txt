conf t

! Set hostname and domain
hostname LAB-SW-01
ip domain-name lab.home

! User and password hardening
enable secret 0 cisco
username soc_admin privilege 15 secret 0 cisco

! Secure remote access (Force SSH, disable Telnet)
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 4
 transport input ssh
 login local
line console 0
 logging synchronous
 login local

! Disable insecure and unnecessary services
no ip http server
no ip http secure-server
no cdp run
vtp mode transparent

! Initialize VLANs
vlan 10
 name VICTIM_NETWORK
vlan 99
 name MANAGEMENT_NETWORK
vlan 999
 name BLACKHOLE
exit

! Configure Switch Management Interface & Routing
interface Vlan99
 ip address 172.16.99.2 255.255.255.0
 no shutdown
exit
ip default-gateway 172.16.99.1

! Disable all unused ports
interface range GigabitEthernet1/0/2 - 26 , GigabitEthernet1/0/29 - 46 , GigabitEthernet1/0/48
 description UNUSED_BLACKHOLE
 switchport mode access
 switchport access vlan 999
 shutdown
exit

! Port 1: Uplink to Juniper Router
interface GigabitEthernet1/0/1
 description UPLINK_TO_JUNIPER
 switchport mode trunk
 switchport nonegotiate
 switchport trunk native vlan 999
 no shutdown
exit

! Port 27: Proxmox Management & Live Traffic (vmbr0)
interface GigabitEthernet1/0/27
 description PROXMOX_MAIN_UPLINK
 switchport mode trunk
 switchport nonegotiate
 spanning-tree portfast trunk
 switchport trunk native vlan 99
 no shutdown
exit

! Port 28: Proxmox SPAN Destination for SecOnion (vmbr1)
interface GigabitEthernet1/0/28
 description SECONION_CAPTURE_NIC
 switchport nonegotiate
 no switchport port-security
 no shutdown
exit

! Port 47: Analyst Station Management
interface GigabitEthernet1/0/47
 description ANALYST_STATION_MGMT
 switchport mode access
 switchport access vlan 99
 switchport nonegotiate
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 1
 switchport port-security violation restrict
 switchport port-security mac-address sticky
 no shutdown
exit

! Corrected SPAN Session: Mirror Victim traffic (VLAN 10) to Port 28
no monitor session 1
monitor session 1 source vlan 10 both
monitor session 1 destination interface GigabitEthernet1/0/27

! Save configuration
do write memory
