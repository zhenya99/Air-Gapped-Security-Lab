index=network_infoblox source=infoblox sourcetype="infoblox:dns" tag=dns 
| eval long_query=if(len(query)>30, query, null())
| where NOT (
)
| bin _time span=1h
| stats count as total_queries, 
    dc(query) as unique_domains, 
    values(long_query) as long_queries 
    by _time, src

| eval long_queries=mvindex(long_queries, 0, 9)
| eventstats avg(unique_domains) as mean_ud, stdev(unique_domains) as stdev_ud
| eval dga_zscore = if(stdev_ud > 0, (unique_domains - mean_ud) / stdev_ud, 0)
| eventstats avg(total_queries) as mean_t, stdev(total_queries) as stdev_t
| eval volume_zscore = if(stdev_t > 0, (total_queries - mean_t) / stdev_t, 0)

| eval threat_category = case(
    dga_zscore > 3 AND volume_zscore > 3, "Critical: Potential DGA + High Volume C2",
    dga_zscore > 3, "High: High Unique Subdomains (Possible DGA / Tunneling)",
    volume_zscore > 3, "Medium: Volume Outlier (Possible Beaconing)",
    1=1, "Normal"
)
| where threat_category != "Normal"
| sort - dga_zscore, volume_zscore
| table _time, threat_category, src, total_queries, unique_domains, dga_zscore, volume_zscore, long_queries
