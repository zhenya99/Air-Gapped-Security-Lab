index=network_infoblox source=infoblox sourcetype="infoblox:dns" tag=dns
| eval raw_query = query
| eval query = lower(query)
| eval query_length = len(query)
| eval long_query = if(query_length > 30, query, null())

| where NOT ()

| eval char_variety = round(len(replace(query, "(.)(?=.*\1)", "")) / query_length, 2)
| eval is_base32 = if(match(query, "(^|\.)[a-z2-7]{16,}(\.|$)"), 1, 0)
| eval is_base64_like = if(match(raw_query, "(^|\.)[a-zA-Z0-9_=-]{20,}(\.|$)") AND match(raw_query, "[A-Z]") AND match(raw_query, "[a-z]") AND match(raw_query, "[0-9]"), 1, 0)

| bin _time span=1h
| stats count as total_queries,
    dc(query) as unique_domains,
    sum(query_length) as total_payload_bytes,
    avg(query_length) as avg_query_len,
    max(query_length) as max_query_len,
    max(char_variety) as max_entropy_ratio,
    sum(is_base32) as base32_matches,
    sum(is_base64_like) as base64_matches,
    values(long_query) as long_queries
    by _time, src

| eval long_queries = mvindex(long_queries, 0, 9)
| eval avg_query_len = round(avg_query_len, 2)

| eventstats avg(unique_domains) as mean_ud, stdev(unique_domains) as stdev_ud
| eval dga_zscore = if(stdev_ud > 0, round((unique_domains - mean_ud) / stdev_ud, 2), 0)

| eventstats avg(avg_query_len) as mean_len, stdev(avg_query_len) as stdev_len
| eval length_zscore = if(stdev_len > 0, round((avg_query_len - mean_len) / stdev_len, 2), 0)

| eval threat_category = case(
    (base32_matches > 5 OR base64_matches > 5) AND length_zscore > 2, "Critical: Confirmed Encoded DNS Exfiltration (Base32/Base64)",
    max_entropy_ratio > 0.60 AND max_query_len > 40, "High: High Entropy / Random Subdomain Payload",
    length_zscore > 3 AND total_payload_bytes > 50000, "High: Excessive Payload Volume Exfiltration",
    dga_zscore > 3, "Medium: Unique Subdomain Anomaly (Possible DGA)",
    1=1, "Normal Traffic"
)
| where threat_category != "Normal Traffic"
| sort - base32_matches, base64_matches, max_entropy_ratio
| table _time, threat_category, src, base32_matches, base64_matches, max_entropy_ratio, total_payload_bytes, avg_query_len, length_zscore, long_queries
